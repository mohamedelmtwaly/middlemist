```sql
-- Create function to replace products in a single transaction
-- This function performs atomic replacement of all products
-- SECURITY: Only authorized users can execute this function
CREATE OR REPLACE FUNCTION replace_products_catalog(new_products JSONB)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  result JSONB;
  inserted_count INTEGER;
  deleted_count INTEGER;
  input_count INTEGER;
  cart_items_count INTEGER;
  order_items_count INTEGER;
  favorites_count INTEGER;
BEGIN
  -- Validate input: must have exactly 348 products
  input_count := jsonb_array_length(new_products);
  IF input_count != 348 THEN
    RAISE EXCEPTION 'Invalid input: expected 348 products, got %', input_count;
  END IF;

  -- Safety check: ensure no related records exist before deletion
  SELECT COUNT(*) INTO cart_items_count FROM cart_items;
  IF cart_items_count > 0 THEN
    RAISE EXCEPTION 'Cannot delete products: cart_items has % related records', cart_items_count;
  END IF;

  SELECT COUNT(*) INTO order_items_count FROM order_items;
  IF order_items_count > 0 THEN
    RAISE EXCEPTION 'Cannot delete products: order_items has % related records', order_items_count;
  END IF;

  SELECT COUNT(*) INTO favorites_count FROM favorites;
  IF favorites_count > 0 THEN
    RAISE EXCEPTION 'Cannot delete products: favorites has % related records', favorites_count;
  END IF;

  -- Delete all existing products from products table only
  DELETE FROM products;
  GET DIAGNOSTICS deleted_count = ROW_COUNT;

  -- Insert new products
  INSERT INTO products (
    name,
    category,
    price,
    image_url,
    description,
    is_active,
    sort_order,
    category_id,
    legacy_id,
    is_featured,
    occasions,
    display_sections,
    subcategory_id
  )
  SELECT
    (elem->>'name')::text,
    (elem->>'category')::text,
    (elem->>'price')::numeric,
    (elem->>'image_url')::text,
    (elem->>'description')::jsonb,
    (elem->>'is_active')::boolean,
    (elem->>'sort_order')::integer,
    (elem->>'category_id')::uuid,
    (elem->>'legacy_id')::text,
    (elem->>'is_featured')::boolean,
    (elem->>'occasions')::jsonb,
    (elem->>'display_sections')::jsonb,
    (elem->>'subcategory_id')::jsonb
  FROM jsonb_array_elements(new_products) AS elem;

  GET DIAGNOSTICS inserted_count = ROW_COUNT;

  -- Verify insert count matches input
  IF inserted_count != 348 THEN
    RAISE EXCEPTION 'Insert mismatch: expected 348, inserted %', inserted_count;
  END IF;

  -- Build success result
  result = jsonb_build_object(
    'success', true,
    'deleted_count', deleted_count,
    'inserted_count', inserted_count,
    'message', 'Products replaced successfully'
  );

  RETURN result;

EXCEPTION
  WHEN OTHERS THEN
    -- Re-raise the exception to trigger automatic rollback
    RAISE;
END;
$$;

-- Grant execute permission to authenticated users only
GRANT EXECUTE ON FUNCTION replace_products_catalog(JSONB) TO authenticated;
```
